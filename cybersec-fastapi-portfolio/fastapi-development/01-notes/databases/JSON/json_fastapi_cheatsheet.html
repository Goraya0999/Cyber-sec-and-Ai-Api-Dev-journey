✓Syntax Rules
  • Quotes required: Keys must use "double" quotes
  • No trailing commas: [1,2,3,] is invalid
  • No comments: JSON has no comment syntax
  • No undefined: Use null instead
  • Escaped characters: \", \\, \n, \t
  • No single quotes: 'string' is invalid JSON

🐍json Module Basics
import json

# Parse JSON string → Python dict
data = json.loads('{"name":"Alice","age":25}')

# Serialize Python dict → JSON string
text = json.dumps(data)
pretty = json.dumps(data, indent=2)

# Read from file
with open("data.json") as f:
    data = json.load(f)

# Write to file
with open("data.json", "w") as f:
    json.dump(data, f, indent=2)
🔧dumps() Options
# Indentation (pretty print)
json.dumps(data, indent=2)

# Sort keys alphabetically
json.dumps(data, sort_keys=True)

# Handle non-JSON types
json.dumps(data, default=str)

# Ensure ASCII only
json.dumps(data, ensure_ascii=False)

# Skip None values
json.dumps(
    {k: v for k, v in data.items() 
     if v is not None}
)
🔄Type Mapping
Python → JSON

  • list, tuple → array
  • str → string
  • int, float → number
  • True → true
  • False → false
  • None → null
JSON → Python
  • object → dict
  • array → list
  • string → str
  • number → int/float
  • true → True
  • false → False
  • null → None
⚠️Common Errors
# ValueError: No JSON object
json.loads('not valid json')

# TypeError: Object of type datetime
datetime.now()  # ← Can't serialize

# Fix with custom encoder
class DateEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, datetime):
            return obj.isoformat()
        return super().default(obj)

json.dumps(data, cls=DateEncoder)

✨Basic Model
from pydantic import BaseModel

class User(BaseModel):
    name: str
    age: int
    email: str | None = None

# Parse JSON
user = User.model_validate_json(
    '{"name":"Bob","age":30}'
)

# Serialize back
json_str = user.model_dump_json()
dict_data = user.model_dump()
🔍Validation
from pydantic import Field, field_validator

class User(BaseModel):
    name: str = Field(
        min_length=1, 
        max_length=100
    )
    age: int = Field(ge=0, le=150)
    email: str = Field(pattern=r"^[\w\.-]+@[\w\.-]+\.\w+$")

    @field_validator('name')
    @classmethod
    def name_uppercase(cls, v):
        return v.upper()

# This fails validation
User(name="", age=200, email="invalid")

📋Config Options
class User(BaseModel):
    model_config = ConfigDict(
        # Include/exclude fields

🎯Field Alias
class User(BaseModel):
    user_name: str = Field(alias="userName")
    first_name: str = Field(alias="firstName")

# Accept either camelCase or snake_case
User(userName="bob", firstName="Bob")
User(user_name="bob", first_name="Bob")

# Serialize with alias
user.model_dump(by_alias=True)
# {"userName": "bob", "firstName": "Bob"}
🔐Exclude/Include
class User(BaseModel):
    name: str
    email: str
    password: str

user = User(
    name="Alice", 
    email="a@example.com",
    password="secret123"
)

# Exclude fields
user.model_dump(exclude={'password'})
# {'name': 'Alice', 'email': 'a@example.com'}

# Include only specific
user.model_dump(include={'name', 'email'})
💡 Pydantic v2 Syntax

Use model_validate_json(), model_dump_json(), model_dump() (not parse_raw(), json(), dict())

🚀Request Body
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class User(BaseModel):
    name: str
    email: str
    age: int

@app.post("/users")
async def create_user(user: User):
    # user is auto-parsed from JSON
    return {"status": "created", "user": user}

# POST {"name":"Bob","email":"b@ex.com","age":30}
# Returns: {"status":"created","user":{...}}

@app.get("/users/{user_id}",

    return {

# Response always matches UserResponse

⚠️ Response Model Matters

Always use response_model for API contracts. It validates output, excludes sensitive fields, and generates OpenAPI docs.

🔀Polymorphic Models
from typing import Literal, Union

class Dog(BaseModel):
    pet_type: Literal["dog"]
    breed: str

class Cat(BaseModel):
    pet_type: Literal["cat"]
    color: str

Pet = Union[Dog, Cat]

@app.post("/pets")
async def add_pet(pet: Pet):
    # Discriminates on pet_type
    return pet

# POST {"pet_type":"dog","breed":"Labrador"}

🔐Partial Updates (PATCH)
class UserUpdate(BaseModel):
    name: str | None = None
    email: str | None = None
    age: int | None = None

@app.patch("/users/{user_id}")
async def update_user(
    user_id: int,
    user_update: UserUpdate
):
    db_user = get_user(user_id)
    update_data = user_update.model_dump(
        exclude_unset=True  # Only set fields
    )
    db_user.update(update_data)
    return db_user

🛡️Custom Validators
from pydantic import field_validator

class User(BaseModel):
    email: str
    
    @field_validator('email')
    @classmethod
    def email_valid(cls, v):
        if '@' not in v:
            raise ValueError('Invalid email')
        return v.lower()

# Invalid → ValidationError
# Valid → Normalized to lowercase

⚙️Custom Error Handler
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request, exc):
    return JSONResponse(
        status_code=422,
        content={
            "error": "Invalid request",
            "details": exc.errors()
        }
    )
